def _parse_price(text):
    if not text:
        return None
    cleaned = re.sub(r"[^\d.]", "", text.replace(",", ""))
    try:
        val = float(cleaned)
        return val if val > 0 else None
    except ValueError:
        return None


def scrape_pepperfry(url):
    try:
        time.sleep(random.uniform(2, 4))
        resp = requests.get(url, headers=HEADERS, timeout=15)
        soup = BeautifulSoup(resp.text, "html.parser")
        offer = None
        for sel in [".pdp-selling-price", ".selling-price", ".special-price",
                    "[data-price]", ".price-box .price"]:
            el = soup.select_one(sel)
            if el:
                offer = _parse_price(el.get_text())
                if offer:
                    break
        mrp = None
        for sel in [".pdp-mrp", ".old-price", ".regular-price", ".price-box .old-price"]:
            el = soup.select_one(sel)
            if el:
                val = _parse_price(el.get_text())
                if val and val != offer:
                    mrp = val
                    break
        return {"mrp": mrp, "offer": offer, "error": None}
    except Exception as e:
        return {"mrp": None, "offer": None, "error": str(e)}

# ── Comparison logic ──────────────────────────────────────────────────────────

def calc_diff_pct(pf_price, ic_offer):
    if pf_price and ic_offer and pf_price > 0:
        return round(((ic_offer - pf_price) / pf_price) * 100, 2)
    return None


def calc_discount_depth(offer, mrp):
    if mrp and offer and mrp > 0:
        return round(((mrp - offer) / mrp) * 100, 1)
    return None


def get_comp_flag(diff_pct):
    if diff_pct is None:
        return "?"
    return "Y" if diff_pct >= 0 else "N"


def get_uncomp_band(diff_pct, comp_flag):
    if diff_pct is None:
        return "N/A"
    if comp_flag == "Y":
        return "Comp&<=2%" if abs(diff_pct) <= 2 else "Comp&>2%"
    abs_diff = abs(diff_pct)
    if abs_diff < 5:
        return "<5%"
    elif abs_diff < 10:
        return "5%-10%"
    else:
        return ">=10%"


def get_margin_flag(pf_depth, ic_depth):
    if pf_depth is None or ic_depth is None:
        return "N/A", None
    gap = round(pf_depth - ic_depth, 1)
    if gap > 5:
        return "Over-discounting", gap
    elif gap < -5:
        return "Under-discounting", gap
    else:
        return "In-line", gap

def run_checks():
    results = []
    timestamp = datetime.now().strftime("%Y-%m-%d %H:%M:%S")

    for sku in SKUS:
        print(f"\nChecking {sku['sku_id']} ...")

        # Scrape Pepperfry live price
        pf_price = sku["pf_price"]
        pf_mrp = sku["pf_mrp"]
        pf_scraped = scrape_pepperfry(sku["pf_url"])
        if pf_scraped["offer"]:
            pf_price = pf_scraped["offer"]
        if pf_scraped["mrp"]:
            pf_mrp = pf_scraped["mrp"]
        print(f"  PF: ₹{pf_price}")

        # Scrape all three competitor platforms
        platform_data = {}
        for platform, url_key, scraper_fn in [
            ("Amazon",    "amazon_url",   scrape_amazon),
            ("Myntra",    "myntra_url",   scrape_myntra),
            ("Tata Cliq", "tatacliq_url", scrape_tatacliq),
        ]:
            url = sku.get(url_key, "")
            if not url:
                platform_data[platform] = {"offer": None, "mrp": None, "error": "No URL"}
                continue
            scraped = scraper_fn(url)
            platform_data[platform] = scraped
            print(f"  {platform}: ₹{scraped['offer']} | error: {scraped['error']}")

        # Build one result row per competitor platform
        for platform, data in platform_data.items():
            ic_offer = data.get("offer")
            ic_mrp = data.get("mrp")
            url = sku.get({"Amazon": "amazon_url", "Myntra": "myntra_url",
                           "Tata Cliq": "tatacliq_url"}[platform], "")

            diff_pct = calc_diff_pct(pf_price, ic_offer)
            comp_flag = get_comp_flag(diff_pct)
            uncomp_band = get_uncomp_band(diff_pct, comp_flag)
            pf_depth = calc_discount_depth(pf_price, pf_mrp)
            ic_depth = calc_discount_depth(ic_offer, ic_mrp)
            margin_flag, margin_gap = get_margin_flag(pf_depth, ic_depth)

            row = {
                "timestamp": timestamp,
                "SKU ID": sku["sku_id"],
                "SKU Title": sku["title"],
                "Brand": sku["brand"],
                "Platform": platform,
                "IC URL": url,
                "PF MRP": pf_mrp or "",
                "PF Price": pf_price or "",
                "PF Discount Depth %": pf_depth or "",
                "IC MRP": ic_mrp or "",
                "IC Offer": ic_offer or "",
                "IC Discount Depth %": ic_depth or "",
                "Price Difference %": diff_pct if diff_pct is not None else "",
                "Discount Gap %": margin_gap if margin_gap is not None else "",
                "Margin Flag": margin_flag,
                "Comp Flag": comp_flag,
                "Uncomp Band": uncomp_band,
                "Scrape Error": data.get("error") or "",
            }
            results.append(row)

    return results


def append_to_log(results):
    file_exists = os.path.isfile(LOG_FILE)
    with open(LOG_FILE, "a", newline="", encoding="utf-8") as f:
        writer = csv.DictWriter(f, fieldnames=results[0].keys())
        if not file_exists:
            writer.writeheader()
        writer.writerows(results)
    print(f"\nLogged {len(results)} rows to {LOG_FILE}")

def build_summary_email(results):
    total_skus = len(set(r["SKU ID"] for r in results))
    alerts = [r for r in results if r["Comp Flag"] == "N" and r.get("IC Offer")]
    alert_skus = len(set(a["SKU ID"] for a in alerts))
    ok_skus = total_skus - alert_skus
    parity_pct = round((ok_skus / total_skus) * 100) if total_skus else 0

    # Competitive gap: avg diff across all rows with valid data
    diffs = [r["Price Difference %"] for r in results
             if isinstance(r.get("Price Difference %"), (int, float))]
    avg_diff = round(sum(diffs) / len(diffs), 1) if diffs else 0
    avg_diff_str = f"+{avg_diff}%" if avg_diff >= 0 else f"{avg_diff}%"
    avg_diff_color = "#1a7f45" if avg_diff >= 0 else "#c0392b"

    # Platform breakdown
    platform_counts = {}
    for a in alerts:
        p = a["Platform"]
        platform_counts[p] = platform_counts.get(p, 0) + 1

    plat_breakdown = " &middot; ".join(
        f'{_platform_badge(p)} {cnt}' for p, cnt in platform_counts.items()
    ) if platform_counts else "None"

    # Per-SKU summary rows (show worst competitor per SKU)
    sku_rows = ""
    seen = {}
    for r in results:
        sid = r["SKU ID"]
        diff = r.get("Price Difference %")
        if not isinstance(diff, (int, float)):
            continue
        if sid not in seen or diff < seen[sid]["diff"]:
            seen[sid] = {"r": r, "diff": diff}

    for sid, entry in seen.items():
        r = entry["r"]
        diff = r["Price Difference %"]
        diff_color = "#1a7f45" if diff >= 0 else "#c0392b"
        diff_str = f"+{diff}%" if diff >= 0 else f"{diff}%"

def send_email(subject, html_body, attach_csv=True):
    if not SENDER_EMAIL or not SENDER_PASSWORD:
        print("Email credentials not set — skipping.")
        return
    msg = MIMEMultipart("alternative")
    msg["Subject"] = subject
    msg["From"] = SENDER_EMAIL
    msg["To"] = ALERT_EMAIL
    msg.attach(MIMEText(html_body, "html"))
    if attach_csv and os.path.isfile(LOG_FILE):
        with open(LOG_FILE, "rb") as f:
            part = MIMEBase("application", "octet-stream")
            part.set_payload(f.read())
            encoders.encode_base64(part)
            part.add_header("Content-Disposition", f'attachment; filename="{LOG_FILE}"')
            msg.attach(part)
    try:
        with smtplib.SMTP_SSL("smtp.gmail.com", 465) as server:
            server.login(SENDER_EMAIL, SENDER_PASSWORD)
            server.sendmail(SENDER_EMAIL, ALERT_EMAIL, msg.as_string())
        print(f"Email sent: {subject}")
    except Exception as e:
        print(f"Failed to send email: {e}")
